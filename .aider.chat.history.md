import { NextResponse } from 'next/server';
import { RecommendationResponse, ErrorResponse } from '@/lib/bourgaeux';

export const runtime = 'edge';

export async function POST(request: Request) {
  try {
    const { prompt } = await request.json();
    
    if (!prompt || typeof prompt !== 'string') {
      return NextResponse.json<ErrorResponse>({
        error: 'Invalid Input',
        message: 'Please provide a valid prompt'
      }, { status: 400 });
    }

    const response = await fetch('https://api.openai.com/v1/chat/completions', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        Authorization: `Bearer ${process.env.OPENAI_API_KEY}`,
      },
      body: JSON.stringify({
        model: 'gpt-4',
        messages: [
          {
            role: 'system',
            content: `You are Bourgaeux, a lifestyle intelligence network. Generate premium recommendations for food, drink, entertainment, and unique experiences. Respond in JSON format matching this structure:
            {
              "recommendations": [
                {
                  "title": "string",
                  "description": "string", 
                  "category": "food|drink|entertainment|experience",
                  "location": "string?",
                  "priceRange": "$|$$|$$$|$$$$?",
                  "bestTime": "string?",
                  "vibe": "string?",
                  "whyUnique": "string?"
                }
              ],
              "summary": "string"
            }`
          },
          {
            role: 'user',
            content: prompt
          }
        ],
        response_format: { type: 'json_object' }
      })
    });

    if (!response.ok) {
      throw new Error('Failed to fetch recommendations');
    }

    const data = await response.json();
    const recommendations = JSON.parse(data.choices[0].message.content);

    return NextResponse.json<RecommendationResponse>(recommendations);
    
  } catch (error) {
    console.error('Recommendation error:', error);
    return NextResponse.json<ErrorResponse>({
      error: 'Recommendation Failed',
      message: 'We encountered an issue generating your recommendations. Please try again.'
    }, { status: 500 });
  }
}
